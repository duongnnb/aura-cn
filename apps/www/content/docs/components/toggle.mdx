## Usage

```tsx
import { AuraToggle } from "@/registry/aura-cn/aura-toggle"

export default function Example() {
  return <AuraToggle defaultChecked />
}
```

## Controlled

```tsx
const [enabled, setEnabled] = useState(false)

<AuraToggle checked={enabled} onCheckedChange={setEnabled} />
```

## Props

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `checked` | `boolean` | — | Controlled state |
| `defaultChecked` | `boolean` | `false` | Initial uncontrolled state |
| `onCheckedChange` | `(checked: boolean) => void` | — | State change callback |
